# ==================== HOME MANAGER MODULE (configuración declarativa) ====================
# Guardar como: ~/.config/nix/awesome-module.nix
# Importar en home.nix: imports = [ ./awesome-module.nix ];
#
# IMPORTANTE: sddm-astronaut-theme NO está en nixpkgs.
# Usa el tema incluido en este repo (sddm/sddm-astronaut-theme) con path local:
#   theme = /ruta/absoluta/a/awesome-rxyhn-remix/sddm/sddm-astronaut-theme;
# O usa un tema de nixpkgs: pkgs.sddm-sugar-candy, pkgs.sddm-chili, etc.

{ config, pkgs, ... }:

let
  myFonts = with pkgs; [
    nerdfonts.jetbrains-mono
    nerdfonts.iosevka
    nerdfonts.hack
    nerdfonts.font-awesome
    nerdfonts.material-design-icons
  ];
  # Tema SDDM incluido en el repo (ajusta la ruta si clonas en otro lado)
  sddmThemePath = "/home/${config.users.users.${config.home-manager.users.${config.home-manager.username}.name}.home}/awesome-rxyhn-remix/sddm/sddm-astronaut-theme";
in
{
  home.packages = with pkgs; [
    awesome
    picom
    kitty
    rofi
    acpi
    acpid
    wireless-tools
    jq
    inotify-tools
    polkit-gnome
    xdotool
    xclip
    maim
    cliphist
    brightnessctl
    alsa-utils
    pipewire
    pipewire-pulse
    wireplumber
    pipewire-alsa
    qt5.qtimageformats
    qt6.qtimageformats
    playerctl
    curl
    ffmpeg
    gpick
    imagemagick
    thunar
    firefox
    xorg.xrdb
    xorg.xrandr
    xorg.setxkbmap
    xorg.xset
    xorg.xprop
    xdg-utils
    xdg-user-dirs
    bc
    upower
    autorandr
    udiskie
    udisks2
    git
    rsync
    spotify
    # libreoffice REMOVIDO (no necesario)
    mpd
    mpc
    ncmpcpp
    blueman
    pasystray
    touchegg
    redshift
    networkmanager
    bluez
    bluez-tools
    libnotify
    sound-theme-freedesktop
    zsh
    zsh-syntax-highlighting
    zsh-autosuggestions
    zoxide
    fzf
    feh
    neovim
    btop
    lsd
    bat
    lua54
    lua54Packages.luarocks
    i3lock
    slock
    xsecurelock
    # Fonts
  ] ++ myFonts;

  # Fonts
  fonts.fonts = with pkgs; myFonts;

  # Services
  services.mpds.enable = true;
  services.udiskie.enable = true;
  services.touchegg.enable = true;
  services.redshift.enable = true;
  services.pipewire = {
    enable = true;
    pulse.enable = true;
    alsa.enable = true;
  };
  services.wireplumber.enable = true;

  # SDDM - usa tema del repo (ajusta sddmThemePath si es necesario)
  services.xserver.displayManager.sddm = {
    enable = true;
    theme = sddmThemePath;
  };

  # Lua PAM
  programs.lua.enable = true;
  programs.lua.package = pkgs.lua54Packages.luapam;
}
